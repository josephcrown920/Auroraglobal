@app.post("/generate")
async def generate_urban_cut(
    model_image: UploadFile = File(...), 
    outfit_image: UploadFile = File(...)
):
    try:
        # Read the raw byte data from the uploaded files
        model_bytes = await model_image.read()
        outfit_bytes = await outfit_image.read()
        
        # 1. Upload files to Fal's content delivery storage to get hosted URLs
        model_url = fal_client.upload(model_bytes, "image/jpeg")
        outfit_url = fal_client.upload(outfit_bytes, "image/jpeg")
        
        # 2. Build the tailored system prompt referencing both assets
        engineered_prompt = (
            f"A vertical 9:16 high-fashion transition video reel. The person from the base image {model_url} "
            f"transitions to wearing the clothing items shown in the target outfit image {outfit_url}. "
            f"Maintain strict facial identity, hair, and body composition from the base image. "
            f"The video includes a sharp, rhythmic camera snap-zoom exactly at the midway frame to replicate a beat-drop. "
            f"Fluid motion, detailed apparel physics, realistic studio tracking shot."
        )
        
        # 3. Queue the generation request to the Luma Dream Machine endpoint on Fal
        # You can also swap this endpoint for 'fal-ai/kling-video' depending on your preference
        result = fal_client.subscribe(
            "fal-ai/luma-dream-machine",
            arguments={
                "prompt": engineered_prompt,
                "image_url": model_url, # Sets the initial starting frame configuration
                "aspect_ratio": "9:16",
                "loop": False
            },
            with_logs=True
        )
        
        # Return the resulting payload containing the video download URL
        return JSONResponse(content={"status": "success", "data": result})
        
    except Exception as e:
        return JSONResponse(status_code=500, content={"error": str(e)})
