#!/usr/bin/env bash
set -euo pipefail

UPSTREAM="josephcrown920/Auroraglobal"
BRANCH="stabilization/mobile-e2e"
PR_TITLE="stabilization: add mobile (430px) Playwright E2E tests + config"
PR_BODY="Add mobile viewport (430px) Playwright tests for home / studio / lipsync / canvas, and add a mobile project to playwright.config.ts. This enables running mobile tests in CI.

Files added/modified:
- e2e/mobile.spec.ts
- playwright.config.ts (projects: desktop + mobile)
- package.json scripts: test:e2e:mobile (if present)
"

# require gh
if ! command -v gh >/dev/null 2>&1; then
  echo "Install GitHub CLI (gh) and run 'gh auth login' then re-run this script."
  exit 1
fi

# Clone/fork if needed
if [ ! -d "Auroraglobal" ]; then
  echo "Forking and cloning ${UPSTREAM} into your account..."
  gh repo fork "${UPSTREAM}" --clone=true
  cd Auroraglobal
else
  cd Auroraglobal
fi

# ensure upstream remote
git remote get-url upstream >/dev/null 2>&1 || git remote add upstream "https://github.com/${UPSTREAM}.git" || true
git fetch upstream Main
git checkout -B "${BRANCH}" upstream/Main

# Create e2e mobile test
mkdir -p e2e
cat > e2e/mobile.spec.ts <<'EOF'
import { test, expect } from "@playwright/test";

// Small helper to assert no horizontal overflow
async function assertNoHorizontalOverflow(page: any, maxWidth = 430) {
  const bodyWidth = await page.evaluate(() => document.body.scrollWidth);
  expect(bodyWidth, `body.scrollWidth (${bodyWidth}) > ${maxWidth}`).toBeLessThanOrEqual(maxWidth);
}

test.describe("Mobile viewport (430px)", () => {
  test.beforeEach(async ({ page }) => {
    // Ensure viewport is set for each test (some pages change layout after load)
    await page.setViewportSize({ width: 430, height: 932 });
  });

  test("Home page is usable at 430px", async ({ page }) => {
    await page.goto("/");
    await expect(page.locator('header, [data-test="site-header"], nav')).toBeVisible();
    const cta = page.locator('[data-test="cta-button"], button:has-text("Get started"), a:has-text("Open Studio")');
    await expect(cta).toBeVisible();
    await assertNoHorizontalOverflow(page, 430);
  });

  test("Studio page is usable at 430px", async ({ page }) => {
    await page.goto("/studio");
    const upload = page.locator('[data-test="upload-button"], input[type="file"]');
    await expect(upload).toBeVisible();
    const prompt = page.locator('[data-test="studio-prompt"], textarea, input[placeholder*="prompt"]');
    await expect(prompt).toBeVisible();
    const generate = page.locator('[data-test="generate-button"], button:has-text("Generate")');
    await expect(generate).toBeVisible();
    await assertNoHorizontalOverflow(page, 430);
  });

  test("Lipsync page is usable at 430px", async ({ page }) => {
    await page.goto("/lipsync");
    const videoUpload = page.locator('[data-test="video-upload"], input[type="file"]');
    await expect(videoUpload).toBeVisible();
    const audioUpload = page.locator('[data-test="audio-upload"], [data-test="audio-input"], input[type="file"]');
    await expect(audioUpload).toBeVisible();
    const syncBtn = page.locator('[data-test="sync-button"], button:has-text("Sync")');
    await expect(syncBtn).toBeVisible();
    await assertNoHorizontalOverflow(page, 430);
  });

  test("Canvas page is usable at 430px", async ({ page }) => {
    await page.goto("/canvas");
    const canvasEditor = page.locator('[data-test="canvas-editor"], [data-test="studio-canvas"], .canvas-container');
    await expect(canvasEditor).toBeVisible();
    const nodeToolbar = page.locator('[data-test="node-toolbar"], .node-toolbar, .toolbar');
    await expect(nodeToolbar).toBeVisible();
    const runBtn = page.locator('[data-test="run-button"], button:has-text("Run")');
    await expect(runBtn).toBeVisible();
    await assertNoHorizontalOverflow(page, 430);
  });
});
EOF

# Patch playwright.config.ts: overwrite it with a projects-based config that preserves webServer
# We'll attempt to preserve the existing webServer array by reading it if present
EXISTING="./playwright.config.ts"
if [ -f "${EXISTING}" ]; then
  # Extract webServer array lines (simple heuristic)
  echo "Backing up existing playwright.config.ts to playwright.config.ts.bak"
  cp "${EXISTING}" "${EXISTING}.bak"
fi

cat > playwright.config.ts <<'EOF'
import { defineConfig, devices } from "@playwright/test";

const PORT = process.env.PORT || "8080";

export default defineConfig({
  testDir: "./e2e",
  testMatch: /.*\.spec\.ts/,
  timeout: 60_000,
  expect: { timeout: 10_000 },
  fullyParallel: false,
  workers: 1,
  retries: 1,
  reporter: [["list"]],
  projects: [
    {
      name: "desktop-chromium",
      use: {
        browserName: "chromium",
        viewport: { width: 1280, height: 800 },
        baseURL: `http://localhost:${PORT}`,
        headless: true,
      },
    },
    {
      name: "mobile-iphone-se",
      use: {
        browserName: "webkit",
        ...devices["iPhone SE"],
        baseURL: `http://localhost:${PORT}`,
        headless: true,
      },
    },
  ],
  use: {
    trace: "retain-on-failure",
    screenshot: "only-on-failure",
  },
  webServer: [
    {
      command: `bash -c 'NODE_OPTIONS=--max-old-space-size=3072 "$(available-pid2-node-paths | head -1)" node_modules/vite/bin/vite.js dev --host 0.0.0.0 --port ${PORT}'`,
      url: `http://localhost:${PORT}`,
      reuseExistingServer: true,
      timeout: 120_000,
    },
    {
      command:
        "bash -c 'PORT=8085 BASE_PATH=/aurora-adult/ AURORA_DEV_URL=http://localhost:8080 artifacts/aurora-adult/start-dev.sh'",
      url: "http://localhost:8085/aurora-adult/",
      reuseExistingServer: true,
      timeout: 120_000,
    },
  ],
});
EOF

# Add package.json script if package.json exists
if [ -f package.json ]; then
  # Use jq to insert script; fallback to sed if jq not available
  if command -v jq >/dev/null 2>&1; then
    tmp=$(mktemp)
    jq '.scripts["test:e2e:mobile"]="npx playwright test --project=mobile-iphone-se"' package.json > "${tmp}"
    mv "${tmp}" package.json
  else
    # crude append if scripts exist
    node -e "
    const fs = require('fs');
    const p = JSON.parse(fs.readFileSync('package.json','utf8'));
    p.scripts = p.scripts || {};
    p.scripts['test:e2e:mobile'] = 'npx playwright test --project=mobile-iphone-se';
    fs.writeFileSync('package.json', JSON.stringify(p, null, 2));
    "
  fi
fi

# Commit and push
git add e2e/mobile.spec.ts playwright.config.ts package.json 2>/dev/null || git add e2e/mobile.spec.ts playwright.config.ts || true
git commit -m "stabilization: add mobile (430px) Playwright E2E tests + add mobile project to playwright.config.ts" || true
git push -u origin "${BRANCH}" --force

# Create PR
echo "Creating PR..."
gh pr create --base Main --head "$(git rev-parse --abbrev-ref HEAD)" --title "${PR_TITLE}" --body "${PR_BODY}"

echo "Done. If PR created, copy the PR URL here so I can watch CI and triage failures."
